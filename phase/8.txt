Phase 8 — Dashboard & Main Window GUI

Build the main application window and dashboard. This must look like a modern,
polished GNOME app. Follow every instruction exactly — do not substitute widget
types, do not guess icon names, do not use GTK3 patterns.

────────────────────────────────────────────
CRITICAL RULES — READ BEFORE WRITING ANY CODE
────────────────────────────────────────────

1. NEVER use Gtk.IconTheme.get_default() — it does not exist in GTK4.
   Use Gtk.IconTheme.get_for_display(Gdk.Display.get_default()) instead.

2. NEVER use widget.get_style_context().add_class() — deprecated in GTK4.
   Use widget.add_css_class('classname') instead.

3. NEVER use Gtk.HBox or Gtk.VBox — they do not exist in GTK4.
   Use Gtk.Box(orientation=Gtk.Orientation.HORIZONTAL) etc.

4. NEVER use Gtk.Label.set_markup() without escaping user data first.
   Use GLib.markup_escape_text() on any dynamic string before markup.

5. NEVER guess icon names. Use ONLY these verified GTK4/GNOME icon names:
   go-next-symbolic
   go-previous-symbolic
   list-add-symbolic
   edit-delete-symbolic
   emblem-ok-symbolic
   dialog-warning-symbolic
   audio-volume-high-symbolic
   preferences-system-symbolic
   view-refresh-symbolic
   weather-clear-symbolic
   document-save-symbolic
   document-open-symbolic
   media-playback-start-symbolic
   media-playback-stop-symbolic
   applications-system-symbolic
   system-shutdown-symbolic
   user-info-symbolic
   open-menu-symbolic
   pan-down-symbolic
   pan-end-symbolic
   pan-start-symbolic
   object-select-symbolic
   edit-clear-symbolic

   For any other concept use a plain colored Gtk.Box as placeholder.
   DO NOT invent icon names — they fail silently and show broken icons.

6. NEVER use deprecated Adw.ActionRow for navigation — use Gtk.ListBoxRow
   with manual content for sidebar items.

7. AdwPreferencesGroup requires being inside AdwPreferencesWindow OR used
   directly inside a plain Gtk.ScrolledWindow + Gtk.Box. Do NOT nest
   AdwPreferencesPage inside AdwApplicationWindow directly.

8. For all CSS: load via Gtk.CssProvider and
   Gtk.StyleContext.add_provider_for_display().
   Use provider.load_from_string() with CSS as a Python string.

9. Adw.ViewStack + Adw.ViewSwitcher is the correct GTK4 pattern for tabbed
   content. Do NOT use Gtk.Notebook.

10. Always set hexpand=True / vexpand=True explicitly.

class WellbeingApplication(Adw.Application):
  application_id = 'com.wellbeingapp.App'

  def __init__(self):
    super().__init__(application_id=self.application_id,
                     flags=Gio.ApplicationFlags.DEFAULT_FLAGS)
    self.connect('activate', self._on_activate)

  def _on_activate(self, app):
    self._load_css()
    win = MainWindow(application=app)
    win.present()

  def _load_css(self):
    css_path = Path(__file__).parent.parent / 'themes' / 'style.css'
    css_content = css_path.read_text()
    provider = Gtk.CssProvider()
    provider.load_from_string(css_content)
    Gtk.StyleContext.add_provider_for_display(
      Gdk.Display.get_default(),
      provider,
      Gtk.STYLE_PROVIDER_PRIORITY_APPLICATION
    )

  def make_prayer_card(self, prayer_name) -> Gtk.Box:
    outer = Gtk.Box(orientation=VERTICAL, css_classes=['card'])
    outer.set_hexpand(True)

    inner = Gtk.Box(orientation=VERTICAL, spacing=6,
                    margin_top=16, margin_bottom=16,
                    margin_start=12, margin_end=12)

    name_lbl   = Gtk.Label(label=prayer_name, css_classes=['heading'], xalign=0)
    time_lbl   = Gtk.Label(css_classes=['title-2'], xalign=0)
    jamah_row  = Gtk.Box(orientation=HORIZONTAL, spacing=4)
    jamah_key  = Gtk.Label(label='Jamah ', css_classes=['caption','dim-label'])
    jamah_val  = Gtk.Label(css_classes=['caption'])
    status_lbl = Gtk.Label(css_classes=['caption'], xalign=0)

    jamah_row.append(jamah_key)
    jamah_row.append(jamah_val)
    inner.append(name_lbl)
    inner.append(time_lbl)
    inner.append(jamah_row)
    inner.append(status_lbl)
    outer.append(inner)

    Store references:
      self._prayer_labels[prayer_name.lower()] = {
        'time': time_lbl,
        'jamah': jamah_val,
        'status': status_lbl,
        'card': outer
      }
    return outer

  def _set_prayer_status(self, prayer_name, status):
    lbl = self._prayer_labels[prayer_name]['status']
    card = self._prayer_labels[prayer_name]['card']

    for cls in ['success', 'accent', 'dim-label']:
      lbl.remove_css_class(cls)
    card.remove_css_class('prayer-card-current')

    if status == 'current':
      lbl.set_label('● Now')
      lbl.add_css_class('success')
      card.add_css_class('prayer-card-current')
    elif status == 'next':
      lbl.set_label('● Next')
      lbl.add_css_class('accent')
    elif status == 'passed':
      lbl.set_label('Passed')
      lbl.add_css_class('dim-label')
    else:
      lbl.set_label('Upcoming')
      lbl.add_css_class('dim-label')

  ── Section 4: Sunrise ──────────────────────

  Gtk.Box(orientation=HORIZONTAL, spacing=8):
    Gtk.Image.new_from_icon_name('weather-clear-symbolic')
    Gtk.Label(label='Sunrise', css_classes=['body'])
    self._sunrise_label = Gtk.Label(css_classes=['body','dim-label'],
                                     hexpand=True, xalign=1)

  ── Data loading ────────────────────────────

  def refresh_data(self):
    Called on init and every 60 seconds via GLib.timeout_add_seconds(60, ...).
    
    schedule = prayer_engine.get_today_schedule()
    display  = prayer_engine.get_schedule_display()
    jamah    = jamah_store.get_all_jamah_times()
    current  = prayer_engine.get_current_prayer()
    next_p, next_t = prayer_engine.get_next_prayer()

    self._date_label.set_label(datetime.now().strftime('%A, %d %B %Y'))
    self._hijri_label.set_label(gregorian_to_hijri(date.today()))
    self._sunrise_label.set_label(display.get('sunrise', '—'))

    self._next_prayer_name.set_label(next_p.capitalize())
    self._next_prayer_time.set_label(f"at {next_t.strftime('%H:%M')}")

    prayer_order = ['fajr', 'dhuhr', 'asr', 'maghrib', 'isha']
    for prayer in prayer_order:
      self._prayer_labels[prayer]['time'].set_label(display.get(prayer, '—'))
      j = jamah.get(prayer)
      self._prayer_labels[prayer]['jamah'].set_label(j if j else '—')

    current_p = prayer_engine.get_current_prayer()
    next_p_name, _ = prayer_engine.get_next_prayer()
    for prayer in prayer_order:
      if prayer == current_p:
        self._set_prayer_status(prayer, 'current')
      elif prayer == next_p_name:
        self._set_prayer_status(prayer, 'next')
      elif schedule[prayer] < datetime.now(schedule[prayer].tzinfo):
        self._set_prayer_status(prayer, 'passed')
      else:
        self._set_prayer_status(prayer, 'upcoming')

  def _update_countdown(self):
    secs = prayer_engine.get_remaining_seconds()
    h = secs // 3600
    m = (secs % 3600) // 60
    s = secs % 60
    if h > 0:
      self._countdown_label.set_label(f'{h}:{m:02d}:{s:02d}')
    else:
      self._countdown_label.set_label(f'{m:02d}:{s:02d}')
    return GLib.SOURCE_CONTINUE

  File browser:
    def _on_browse(self, btn, prayer, file_label):
      dialog = Gtk.FileDialog.new()
      dialog.set_title('Select Adhan Audio')
      filter_ = Gtk.FileFilter()
      filter_.set_name('Audio Files')
      filter_.add_mime_type('audio/mpeg')
      filter_.add_mime_type('audio/ogg')
      filter_.add_mime_type('audio/wav')
      filters = Gio.ListStore.new(Gtk.FileFilter)
      filters.append(filter_)
      dialog.set_filters(filters)
      dialog.open(self.get_root(), None, self._on_file_chosen, prayer, file_label)

        file = dialog.open_finish(result)
        path = file.get_path()
        file_label.set_label(os.path.basename(path))
        self._save_audio_path(prayer, path)
      except GLib.Error:
        pass
