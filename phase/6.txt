  def get_overlay_widgets(self, slot: str) -> list:
    import json
    rows = self.conn.execute('''
      SELECT wt.id, wt.widget_type, wt.config_json, owa.display_order
      FROM overlay_widget_assignments owa
      JOIN widget_templates wt ON wt.id = owa.widget_template_id
      WHERE owa.overlay_slot = ? AND owa.enabled = 1
      ORDER BY owa.display_order ASC
    ''', (slot,)).fetchall()

    return [
      WidgetConfig(
        template_id=row[0],
        widget_type=row[1],
        config=json.loads(row[2] or '{}'),
        display_order=row[3]
      )
      for row in rows
    ]

class OverlayWindow(Gtk.Window, GObject.GObject):

  __gsignals__ = {
    'overlay-dismissed': (
      GObject.SignalFlags.RUN_FIRST,
      None,
      (str,)
    )
  }

  Dismissed reasons (always pass one of these strings to emit):
    'timer_expired_keypress'   → lock ended, user pressed key
    'timer_expired_timeout'    → lock ended, user did nothing for 5s
    'escape'                   → user dismissed unlocked overlay
    'password_accepted'        → correct password entered

  def __init__(self, config: OverlayConfig):
    ...

  def _dismiss(self, reason: str):
    self.emit('overlay-dismissed', reason)
    self.destroy()

  Center card: glassmorphism rounded rect
    CSS class 'overlay-card':

  Fade-in: opacity 0→1 over 400ms using GLib.timeout_add(16, step)

LOCKED mode behavior:
  Phase 1 (locked, 0 to duration_seconds):
    Key events do nothing (controller connected but ignored)
    Show countdown timer
    If password set: show PIN entry (always visible), Enter checks password
  
  Phase 2 (unlockable, after timer):
    Show 'Press Enter or Space to continue'
    Enter / Space → self._dismiss('timer_expired_keypress')
    If no key in 5 seconds → self._dismiss('timer_expired_timeout')
    If password: Enter checks PIN,
      correct → self._dismiss('password_accepted')

UNLOCKED mode behavior:
  NOT keyboard-grabbing (no EXCLUSIVE keyboard mode on layer shell)
  Escape / Space / Enter → self._dismiss('escape')
  If auto_close=True: after duration_seconds → self._dismiss('timer_expired_timeout')

class OverlayManager:
  _instance = None

  @classmethod
  def get(cls):
    if cls._instance is None:
      cls._instance = OverlayManager()
    return cls._instance

  def __init__(self):
    self._current: OverlayWindow | None = None
    self._queue: list[OverlayConfig] = []
    self._callbacks: list = []

  def show_overlay(self, config: OverlayConfig):
    if self._current is not None:
      self._queue.append(config)
      return
    self._show(config)

  def _show(self, config: OverlayConfig):
    win = OverlayWindow(config)
    win.connect('overlay-dismissed', self._on_dismissed)
    self._current = win
    win.present()

  def _on_dismissed(self, window, dismissed_by: str):
    config = self._current.config if self._current else None
    self._current = None
    for cb in self._callbacks:
      cb(config, dismissed_by)
    if self._queue:
      next_config = self._queue.pop(0)
      self._show(next_config)

  def on_dismissed(self, callback):
    self._callbacks.append(callback)

  def dismiss_current(self):
    if self._current:
      self._current._dismiss('escape')

  def is_overlay_active(self) -> bool:
    return self._current is not None

Connect daemon events to OverlayManager in daemon/service.py:

  overlay_manager = OverlayManager.get()
  overlay_manager.on_dismissed(self._on_overlay_dismissed)

  def _on_overlay_dismissed(self, config, dismissed_by: str):
    if config and config.overlay_id == 'eyecare':
      if dismissed_by == 'timer_expired_keypress':
        self._fire_zikr_overlay()

  def _fire_lock_overlay(self, prayer_name: str):
    widgets = db.get_overlay_widgets('salah_lock')
    config = OverlayConfig(
      mode='locked',
      duration_seconds=int(db.get_config('salah_lock_duration_seconds', 900)),
      widgets=widgets,
      overlay_id='salah_lock'
    )
    overlay_manager.show_overlay(config)

  def _fire_zikr_overlay(self):
    if overlay_manager.is_overlay_active():
      return
    widgets = db.get_overlay_widgets('zikr')
    config = OverlayConfig(
      mode='unlocked',
      duration_seconds=0,
      widgets=widgets,
      overlay_id='zikr'
    )
    overlay_manager.show_overlay(config)
