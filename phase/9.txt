class ZikrSequencePlayer:

  def __init__(self, card_container: Gtk.Box):
    self.container = card_container
    self._sequence = None
    self._steps = []
    self._current_step = 0
    self._timer_id = None
    self._step_dots = []

  def load_sequence(self, sequence_id: int, db):
    seq_row = db.conn.execute(
      'SELECT * FROM zikr_sequences WHERE id=?', (sequence_id,)
    ).fetchone()
    step_rows = db.conn.execute(
      'SELECT * FROM zikr_steps WHERE sequence_id=? ORDER BY step_order',
      (sequence_id,)
    ).fetchall()
    self._sequence = seq_row
    self._steps = step_rows

  def play(self):
    if not self._steps:
      return
    self._current_step = 0
    self._show_step(0)

  def _select_widgets(self, step) -> list[int]:
    import json, random
    widget_ids = json.loads(step['widget_ids'] or '[]')
    mode = step['selection_mode']
    if mode == 'fixed':
      return widget_ids
    elif mode == 'random_one':
      return [random.choice(widget_ids)] if widget_ids else []
    elif mode == 'random_n':
      n = step['n_count'] or 1
      return random.sample(widget_ids, min(n, len(widget_ids)))
    return widget_ids

  def _show_step(self, step_index: int):
    step = self._steps[step_index]
    selected_ids = self._select_widgets(step)

    ← Clear container
    child = self.container.get_first_child()
    while child:
      next_child = child.get_next_sibling()
      self.container.remove(child)
      child = next_child

    ← Load widgets for this step
    for template_id in selected_ids:
      wc = WidgetConfig(
        template_id=template_id,
        widget_type=...,  ← look up from widget_templates
        config=...,
        display_order=0
      )
      widget = WidgetRenderer.create_widget(wc)
      self.container.append(widget)
      if hasattr(widget, 'start_animations'):
        widget.start_animations()

    ← Update step indicator dots
    self._update_dots(step_index)

    ← Schedule advance
    duration = step['duration_seconds'] or 30
    self._timer_id = GLib.timeout_add_seconds(
      duration, self._advance_step
    )

  def _advance_step(self):
    self._current_step += 1
    if self._current_step >= len(self._steps):
      loop_mode = self._sequence['loop_mode'] if self._sequence else 'once'
      if loop_mode == 'loop':
        self._current_step = 0
        self._show_step(0)
      else:
        self.emit_sequence_complete()
      return GLib.SOURCE_REMOVE

    transition = self._steps[self._current_step - 1]['transition']
    if transition == 'fade':
      self._fade_then_show(self._current_step)
    else:
      self._show_step(self._current_step)
    return GLib.SOURCE_REMOVE

  def _fade_then_show(self, step_index: int):
    step_count = [0]
    def fade_out():
      step_count[0] += 1
      opacity = 1.0 - (step_count[0] / 20)
      self.container.set_opacity(max(0.0, opacity))
      if step_count[0] >= 20:
        self._show_step(step_index)
        fade_in_count = [0]
        def fade_in():
          fade_in_count[0] += 1
          self.container.set_opacity(min(1.0, fade_in_count[0] / 20))
          return fade_in_count[0] < 20
        GLib.timeout_add(15, fade_in)
        return False
      return True
    GLib.timeout_add(15, fade_out)

  def _update_dots(self, current: int):
    ← Step indicator: add dots box below container's parent if not exists
    ← Show filled dot for current step, empty for others
    ← Implementation: find or create a Gtk.Box of Gtk.Label dots
    total = len(self._steps)
    dots_text = ' '.join('●' if i == current else '○' for i in range(total))
    if not hasattr(self, '_dots_label'):
      self._dots_label = Gtk.Label(css_classes=['caption', 'dim-label'])
      self._dots_label.set_halign(Gtk.Align.CENTER)
      self.container.append(self._dots_label)
    self._dots_label.set_label(dots_text)

  def emit_sequence_complete(self):
    ← Signal parent OverlayWindow to close
    ← Walk up widget tree to find OverlayWindow and call _dismiss
    parent = self.container.get_parent()
    while parent:
      if isinstance(parent, OverlayWindow):
        parent._dismiss('sequence_complete')
        return
      parent = parent.get_parent()

  def stop(self):
    if self._timer_id:
      GLib.source_remove(self._timer_id)
      self._timer_id = None
